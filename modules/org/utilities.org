#+begin_src emacs-lisp :tangle yes
(defun my/tangle-on-save ()
  "Tangle Org files before saving."
  (when (and buffer-file-name
             (string-match-p "\\.org$" buffer-file-name)
             (string-prefix-p (expand-file-name "~/.config/doom/modules/")
                              (expand-file-name buffer-file-name)))
    (org-babel-tangle)
    (message "🔁 Tangled %s" buffer-file-name)))
(add-hook 'before-save-hook #'my/tangle-on-save)
#+end_src

** Zathura Integration

Focused configuration: Defines a custom command or function.

#+begin_src emacs-lisp :tangle yes
(defun my/open-pdf-in-zathura ()
  "Open the current PDF in Zathura."
  (interactive)
  (let ((file (buffer-file-name)))
    (start-process "zathura" nil "zathura" file)))

(defun my/launch-zathura-prompt ()
  "Prompt for a PDF file and open it in Zathura."
  (interactive)
  (let ((file (read-file-name "Open PDF in Zathura: " nil nil t nil
                              (lambda (f) (string-match-p "\\.pdf\\'" f)))))
    (start-process "zathura" nil "zathura" (expand-file-name file))))

(defun my/open-pdf-in-zathura-and-close ()
  "Open the current PDF in Zathura and close the buffer in Emacs."
  (interactive)
  (if (not (buffer-file-name))
      (message "This buffer is not visiting a file.")
    (start-process "zathura" nil "zathura" (buffer-file-name))
    (kill-buffer (current-buffer))))

(defun my/consult-pdf-zathura ()
  "Prompt for directory, then fuzzy search PDFs and open in Zathura."
  (interactive)
  (require 'consult)
  (let* ((dir (read-directory-name "Select directory: " default-directory))
         (file (consult--read
                (directory-files-recursively dir "\\.pdf$")
                :prompt "Open PDF in Zathura: ")))
    (start-process "zathura" nil "zathura" (expand-file-name file))))
#+end_src

** Treemacs and Minimap

Focused configuration: Defines a custom command or function.

#+begin_src emacs-lisp :tangle yes
(defun my/find-file-with-sidebars ()
  "Open a file, then show Minimap and Treemacs focused on the file."
  (interactive)
  (require 'treemacs)
  (let ((file (read-file-name "Find file: ")))
    (find-file file)
    (when (fboundp 'minimap-mode)
      (minimap-mode 1))
    (unless (treemacs-get-local-window)
      (treemacs))
    (treemacs-find-file)
    (other-window 1)))

(defun my/open-all-files-in-directory (dir)
  "Open all files in DIR, show minimap, and open Treemacs."
  (interactive "DDirectory: ")
  (let* ((files (seq-filter #'file-regular-p
                            (directory-files dir t "^[^.].*"))))
    (unless files
      (user-error "No files to open in directory: %s" dir))
    (dolist (file files)
      (find-file file))
    (minimap-mode 1)
    (require 'treemacs)
    (treemacs)
    (other-window 1)
    (run-at-time 0.15 nil #'my/open-sidebars)
    (message "Opened %d files, enabled minimap, and opened Treemacs." (length files))))
#+end_src

** Treemacs Manual Directory Load

Focused configuration: Sets various configuration variables.

#+begin_src emacs-lisp :tangle yes
(defun my/treemacs-open-at-dir (dir &optional no-persist)
  "Open Treemacs at DIR without project logic. Set NO-PERSIST to disable state restore temporarily."
  (interactive "DDirectory: 
P")
  (require 'treemacs)
  (when no-persist
    (setq treemacs--projects nil
          treemacs--workspaces nil
          treemacs--ready-to-follow nil
          treemacs-last-error-persisted nil))
  (let ((default-directory dir)
        (treemacs-no-persist no-persist))
    ;; Kill existing Treemacs buffers
    (dolist (buf (buffer-list))
      (when (string-match-p "\*Treemacs.*\*" (buffer-name buf))
        (kill-buffer buf)))
    ;; Open Treemacs manually
    (treemacs)
    (when no-persist
      (treemacs--reset))
    (treemacs-do-add-project-to-workspace dir (file-name-nondirectory (directory-file-name dir)))
    (treemacs-select-window)))
#+end_src

* Buffer Management

#+begin_src emacs-lisp :tangle yes
(defun my/kill-all-buffers-and-close-sidebars ()
  "Kill all buffers and close Treemacs and Minimap if they are open."
  (interactive)
  (when (y-or-n-p "Really kill ALL buffers and close sidebars? ")
    (+evil:kill-all-buffers)
    (when (featurep 'treemacs)
      (dolist (win (window-list))
        (when (string-match-p "\\*Treemacs" (buffer-name (window-buffer win)))
          (delete-window win)))
      (dolist (buf (buffer-list))
        (when (string-match-p "\\*Treemacs" (buffer-name buf))
          (kill-buffer buf))))
    (when (bound-and-true-p minimap-mode)
      (minimap-mode -1))
    (message "All buffers killed and sidebars closed.")))

(defun my/kill-all-other-buffers ()
  "Kill all buffers except the current one and Treemacs buffers."
  (interactive)
  (when (y-or-n-p "Kill all OTHER buffers (keep Treemacs)? ")
    (mapc (lambda (b)
            (unless (or (eq b (current-buffer))
                        (string-match-p "\\*Treemacs" (buffer-name b)))
              (kill-buffer b)))
          (buffer-list))
    (message "Other buffers killed, Treemacs preserved.")))

(defun my/kill-unmodified-buffers ()
  "Kill all unmodified buffers."
  (interactive)
  (when (y-or-n-p "Kill all unmodified buffers?")
    (dolist (buf (buffer-list))
      (with-current-buffer buf
        (unless (buffer-modified-p)
          (kill-buffer buf))))
    (message "Unmodified buffers killed.")))
#+end_src

  "Rotate themes and apply new one by DIRECTION steps."
  (interactive "p")
