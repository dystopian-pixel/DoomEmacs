* Doom Emacs Interface

This section is the hub for how the editor looks and feels. It focuses on clarity, calmness, and visual consistency so the interface fades into the background while you work. The aim is to make information legible at a glance without visual clutter. It ties together typography, colors, spacing, and subtle cues so the whole environment feels intentional.

** Fonts and Theme

This subsection shapes the overall personality of the interface. Font choices guide readability and rhythm, while the theme sets contrast, warmth, and mood. The goal is pleasant long-form reading with enough pop for quick scanning. It should feel cohesive across different file types and workflows.

#+begin_src emacs-lisp
;;; ui.el --- UI settings -*- lexical-binding: t; -*-

(defvar my/last-theme-file
  (expand-file-name "modules/config/.last-theme.el" doom-user-dir)
  "File used to persist the last selected theme.")

(defun my/write-last-theme-file (theme)
  "Write THEME to `my/last-theme-file'."
  (with-temp-file my/last-theme-file
    (insert (format "(setq doom-theme '%S)\n" theme))))

(defun my/persist-theme-a (theme &rest _)
  "Persist THEME whenever it is enabled or loaded."
  (when (and (symbolp theme) (custom-theme-p theme))
    (setq doom-theme theme)
    (my/write-last-theme-file theme)))

;; Restore the saved theme before Doom applies it at startup.
(if (file-exists-p my/last-theme-file)
    (load my/last-theme-file nil t)
  (setq doom-theme 'doom-one))

;; Capture theme changes from consult-theme, load-theme, dashboard cycling,
;; browser gallery switching, and anything else that enables a theme.
(unless (advice-member-p #'my/persist-theme-a #'load-theme)
  (advice-add 'load-theme :after #'my/persist-theme-a))
(unless (advice-member-p #'my/persist-theme-a #'enable-theme)
  (advice-add 'enable-theme :after #'my/persist-theme-a))

(use-package! doom-themes
  :config
  (setq doom-themes-enable-bold t
        doom-themes-enable-italic t))

(add-to-list 'custom-theme-load-path (expand-file-name "themes" doom-user-dir))

(setq doom-font (font-spec :family "JetBrains Mono" :size 12)
      doom-variable-pitch-font (font-spec :family "Inter" :size 16)
      doom-big-font (font-spec :family "JetBrains Mono" :size 20)
      display-line-numbers-type t)


(defun my/doom-color-or (name &optional fallback)
  "Return Doom theme color NAME, FALLBACK, or `unspecified'."
  (or (and (fboundp 'doom-color)
           (ignore-errors (doom-color name)))
      fallback
      'unspecified))

(defun my/doom-lighten-or (name amount &optional fallback)
  "Return Doom color NAME lightened by AMOUNT, FALLBACK, or `unspecified'."
  (or (and (fboundp 'doom-color)
           (fboundp 'doom-lighten)
           (ignore-errors (doom-lighten (doom-color name) amount)))
      fallback
      'unspecified))

(defun my/doom-blend-or (fg bg alpha &optional fallback)
  "Return Doom color FG blended over BG by ALPHA, FALLBACK, or `unspecified'."
  (or (and (fboundp 'doom-color)
           (fboundp 'doom-blend)
           (ignore-errors (doom-blend (doom-color fg) (doom-color bg) alpha)))
      fallback
      'unspecified))

(custom-set-faces!
  '(font-lock-comment-face :slant italic)
  '(font-lock-comment-delimiter-face :slant italic))


#+end_src

** Line Numbers

Line numbers act like street addresses for your text. They’re there when you need orientation and invisible when you don’t. The emphasis is on quiet contrast that doesn’t compete with the content. They help you coordinate with tools, notes, and collaborators without breaking focus.

#+begin_src emacs-lisp
;; Theme-aware, slightly brighter line numbers.
;; Applies after themes load so it isn't clobbered.
(after! doom-themes
  (custom-set-faces!
   ;; Non-current lines: a bit brighter than the usual gutter
   `(line-number
     :inherit default
     :height 0.95
     :foreground ,(or
                   (and (fboundp 'doom-color) (doom-color 'base7))
                   (and (fboundp 'doom-color) (doom-lighten (doom-color 'base6) 0.18))
                   "#a9aeb8"))
   ;; Current line: bright and bold for focus
   `(line-number-current-line
     :inherit default
     :height 1.0
     :weight bold
     :foreground ,(or
                   (and (fboundp 'doom-color) (doom-lighten (doom-color 'fg) 0.08))
                   "#ffffff"))))
#+end_src

** Column Guide & VC Gutter

The column guide provides a gentle edge to help you keep comfortable line widths. The version-control gutter offers just-enough hints about changes without demanding attention. Together, they support disciplined layout and awareness of edits. The visual tone stays soft so content remains the star.

#+begin_src emacs-lisp
;; Soft, theme-aware fill-column indicator & VC gutter colors
(after! doom-themes
  (let* ((fci (or
               (and (fboundp 'doom-blend) (fboundp 'doom-color)
                    ;; faint line: 25% fg blended over bg
                    (doom-blend (doom-color 'fg) (doom-color 'bg) 0.25))
               (and (fboundp 'doom-color) (doom-lighten (doom-color 'base6) 0.25))
               "#6e7380"))
         (chg (or (and (fboundp 'doom-color) (doom-lighten (doom-color 'cyan)  0.08)) "#7fb4ca"))
         (ins (or (and (fboundp 'doom-color) (doom-lighten (doom-color 'green) 0.08)) "#86c991"))
         (del (or (and (fboundp 'doom-color) (doom-lighten (doom-color 'red)   0.05)) "#d18f96")))
    (custom-set-faces!
     ;; Fill-column indicator: quiet, readable, transparency-friendly
     `(fill-column-indicator
       :inherit shadow
       :foreground ,fci)

     ;; diff-hl (VC gutter): soft tints; keep backgrounds transparent
     `(diff-hl-change :background unspecified :foreground ,chg)
     `(diff-hl-insert :background unspecified :foreground ,ins)
     `(diff-hl-delete :background unspecified :foreground ,del)

     ;; If you use diff-hl in the margin instead of fringe, style those too:
     `(diff-hl-margin-change :background unspecified :foreground ,chg)
     `(diff-hl-margin-insert :background unspecified :foreground ,ins)
     `(diff-hl-margin-delete :background unspecified :foreground ,del))))
#+end_src


** Selection — Theme-Aware
Selections are designed to be unmistakable without feeling heavy. Visibility remains solid even in unusual backgrounds or transparent setups. The aim is clarity when cutting, copying, or refactoring text. You should never wonder what’s selected—or lose the text beneath the highlight.

#+begin_src emacs-lisp
;; Theme-aware selection (region) styling
(after! doom-themes
  (let* ((bg   (and (fboundp 'doom-color) (doom-color 'bg)))
         ;; Prefer the theme's own selection tint; fall back to blue/magenta.
         (acc  (or (and (fboundp 'doom-color) (ignore-errors (doom-color 'selection)))
                   (and (fboundp 'doom-color) (ignore-errors (doom-color 'blue)))
                   (and (fboundp 'doom-color) (ignore-errors (doom-color 'magenta)))))
         ;; Blend accent into the background for a soft highlight.
         ;; Increase the 0.40 → 0.48 if you want a stronger selection.
         (sel-bg (or (and (fboundp 'doom-blend) acc bg (doom-blend acc bg 0.40))
                     (and (fboundp 'doom-lighten) acc (doom-lighten acc 0.25))
                     "#3a5fcd"))      ; conservative fallback
         ;; Keep foreground generally theme-driven; provide a bright fallback.
         (sel-fg (or (and (fboundp 'doom-color) (ignore-errors (doom-color 'fg)))
                     "#ffffff")))
    (custom-set-faces!
     ;; Primary selection
     `(region :background ,sel-bg :foreground ,sel-fg :extend t)
     ;; Secondary selection (used by some modes/commands)
     `(secondary-selection
       :background ,(or (and (fboundp 'doom-blend) sel-bg bg (doom-blend sel-bg bg 0.20))
                        "#4a6fdc")
       :foreground ,sel-fg :extend t))))
#+end_src

* Centaur Tabs

Tabs provide a familiar, visual way to juggle multiple buffers. They serve as labeled anchors so you can hop among contexts with minimal thought. Grouping and order help you keep threads of work tidy. The look aims for clarity over flair, keeping attention on content.

#+begin_src emacs-lisp
(when (modulep! :ui tabs)
  (after! centaur-tabs
    (setq centaur-tabs-style "bar"
          centaur-tabs-height 28
          centaur-tabs-set-icons t
          centaur-tabs-set-bar 'over
          centaur-tabs-show-navigation-buttons t
          centaur-tabs-set-modified-marker t
          centaur-tabs-adjust-buffer-order t
          centaur-tabs-cycle-scope 'tabs)

    (defun my/centaur-tabs-buffer-groups ()
      "Return the current buffer group for Centaur Tabs."
      (list
       (cond
        ((bound-and-true-p my/centaur-tabs-group)
         my/centaur-tabs-group)
        ((derived-mode-p 'org-mode)
         "Org")
        ((derived-mode-p 'emacs-lisp-mode 'lisp-interaction-mode)
         "Elisp")
        ((derived-mode-p 'dired-mode)
         "Dired")
        ((derived-mode-p 'help-mode 'helpful-mode 'special-mode)
         "System")
        ((derived-mode-p 'term-mode 'eshell-mode 'vterm-mode 'shell-mode)
         "Terminal")
        ((and (fboundp 'projectile-project-p)
              (projectile-project-p))
         (format "Project:%s" (projectile-project-name)))
        (t
         "Common"))))

    (setq centaur-tabs-buffer-groups-function #'my/centaur-tabs-buffer-groups)

    (defun my/centaur-tabs-enabled-p ()
      "Return non-nil when Centaur Tabs is really enabled."
      (and (fboundp 'centaur-tabs-mode-on-p)
           (centaur-tabs-mode-on-p)))

    (defun my/centaur-tabs-enable-h ()
      "Enable Centaur Tabs without toggling it off accidentally."
      (unless (my/centaur-tabs-enabled-p)
        (centaur-tabs-mode +1)))

    (defun my/centaur-tabs-disable-on-dashboard-h ()
      "Hide Centaur Tabs locally in Doom dashboard buffers."
      (when (my/centaur-tabs-enabled-p)
        (centaur-tabs-local-mode +1)))

    ;; Enable Centaur Tabs immediately. Do not rely only on theme hooks.
    (my/centaur-tabs-enable-h)

    ;; Keep it enabled after theme changes.
    (add-hook 'doom-load-theme-hook #'my/centaur-tabs-enable-h)

    ;; Hide Centaur Tabs on Doom dashboard buffers only.
    (add-hook '+doom-dashboard-mode-hook #'my/centaur-tabs-disable-on-dashboard-h)
    (add-hook '+dashboard-mode-hook #'my/centaur-tabs-disable-on-dashboard-h)

    ;; If the dashboard already exists, apply the local hide rule now.
    (dolist (buffer (buffer-list))
      (with-current-buffer buffer
        (when (derived-mode-p '+doom-dashboard-mode '+dashboard-mode)
          (my/centaur-tabs-disable-on-dashboard-h))))))
#+end_src

* Theme Selector

This provides a smooth way to try on different visual moods. Switching is quick so you can choose what fits the time of day or task. Previews help you decide without commitment. The intent is exploration without friction.

#+begin_src emacs-lisp
(defun my/theme-choose ()
  "Prompt for and load a curated Doom theme."
  (interactive)
  (let* ((themes '(doom-one doom-dracula doom-nord doom-moonlight doom-gruvbox))
         (choice (intern (completing-read "Theme: " (mapcar #'symbol-name themes)))))
    (setq doom-theme choice)
    (doom/reload-theme)))
#+end_src
