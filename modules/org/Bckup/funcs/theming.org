** Save Selected Theme

#+begin_src emacs-lisp
(defun my/set-doom-theme (theme)
  "Load THEME, persist it, and refresh visuals."
  (interactive
   (list (intern (completing-read "Theme: " (custom-available-themes)))))
  ;; Unload existing themes
  (mapc #'disable-theme custom-enabled-themes)
  ;; Load new theme
  (load-theme theme t)
  ;; Save selected theme
  (with-temp-file (expand-file-name ".last-theme.el" doom-user-dir)
    (insert (format "(setq doom-theme '%S)" theme)))
  ;; Reload visuals
  (doom/reload-theme))

#+end_src


** Persist Selected Theme

#+begin_src emacs-lisp
(defun my/save-doom-theme (theme &rest _)
  "Persist the selected THEME to be loaded next time Doom starts."
  (with-temp-file (expand-file-name ".last-theme.el" doom-user-dir)
    (insert (format "(setq doom-theme '%S)" theme))))

(advice-add 'load-theme :after #'my/save-doom-theme)

#+end_src


** Theme Selector

This section lets you try on different visual moods with minimal effort. Switching is fast so you can match lighting, task, or preference. It encourages exploration while keeping decisions reversible. You pick what feels right in the moment and move on.


#+begin_src emacs-lisp
(defun my/theme-at-index ()
  "Return the theme at current index."
  (interactive)
  (message "%s" (nth my/current-theme-index my/doom-themes-list))
  (nth my/current-theme-index my/doom-themes-list))

(defun my/rotate-theme (direction)
  "Rotate themes and apply new one by DIRECTION steps."
  (interactive "p")
  (setq my/current-theme-index
        (mod (+ my/current-theme-index direction) (length my/doom-themes-list)))
  (my/apply-theme (my/theme-at-index)))
#+end_src
